// colver-core/src/belief/belief_state.rs
//! Persistent belief state that accumulates constraints from bidding through play.
//! Produces determinizations consistent with all observed actions.

use crate::bid_eval::evaluate_for_trump;
use crate::card::*;
use crate::state::GameState;

/// Map encoded bid value (8=80, 9=90, ..., 13=130) to evaluate_for_trump threshold.
/// Inverse of `score_to_bid_value` in eval_helpers.rs.
fn bid_value_to_threshold(bid_value: u8) -> u16 {
    match bid_value {
        0..=8 => 10,   // 80
        9 => 12,        // 90
        10 => 14,       // 100
        11 => 17,       // 110
        12 => 20,       // 120
        _ => 23,        // 130+
    }
}

/// A constraint learned from a single observed action.
#[derive(Clone, Debug)]
pub struct ActionConstraint {
    pub player: u8,
    pub kind: ConstraintKind,
}

/// What kind of constraint was inferred.
#[derive(Clone, Debug)]
pub enum ConstraintKind {
    /// Player bid `suit` at a level implying evaluate_for_trump >= min_score.
    Bid { suit: Suit, min_score: u16 },
    /// Player passed. Context determines how restrictive.
    Pass {
        /// Minimum bid value (encoded) they would have needed to overbid. 0 = opening pass.
        min_overbid_value: u8,
        /// Position in the auction round (0-3).
        auction_position: u8,
        /// Whether partner had already bid.
        partner_had_bid: bool,
        /// The active bid suit, if any.
        active_suit: Option<Suit>,
    },
}

impl ActionConstraint {
    /// Check if a set of hands satisfies this constraint.
    pub fn is_satisfied(&self, hands: &[CardSet; 4]) -> bool {
        let hand = hands[self.player as usize];
        match &self.kind {
            ConstraintKind::Bid { suit, min_score } => {
                evaluate_for_trump(hand, *suit) >= *min_score
            }
            ConstraintKind::Pass {
                min_overbid_value,
                auction_position,
                partner_had_bid,
                active_suit,
            } => {
                if *min_overbid_value == 0 {
                    // Opening pass: no J+9 in any suit
                    let has_j9 = (0..4u8).any(|s| {
                        let bits = suit_bits(hand, Suit::from_u8(s));
                        (bits & (1 << 3) != 0) && (bits & (1 << 2) != 0)
                    });
                    if has_j9 {
                        return false;
                    }
                    // Position 3+ pass is more informative (threshold 8 vs 10)
                    let threshold = if *auction_position >= 2 { 8 } else { 10 };
                    (0..4u8).all(|s| evaluate_for_trump(hand, Suit::from_u8(s)) < threshold)
                } else {
                    let overbid_threshold = bid_value_to_threshold(*min_overbid_value);
                    if *partner_had_bid {
                        // Can't support partner's suit
                        if let Some(suit) = active_suit {
                            evaluate_for_trump(hand, *suit) < overbid_threshold
                        } else {
                            true
                        }
                    } else {
                        // Can't overbid in any suit
                        (0..4u8)
                            .all(|s| evaluate_for_trump(hand, Suit::from_u8(s)) < overbid_threshold)
                    }
                }
            }
        }
    }
}

/// Persistent belief state accumulating constraints from bidding through play.
pub struct BeliefState {
    pub observer: u8,
    observer_hand: CardSet,
    /// Soft weights for biased sampling (multiplicative).
    pub soft_weights: [[f32; 32]; 4],
    /// Hard constraints accumulated from observed actions.
    constraints: Vec<ActionConstraint>,
}

impl BeliefState {
    /// Create a new BeliefState for the given observer.
    pub fn new(observer: u8, observer_hand: CardSet) -> Self {
        // Initialize soft weights: observer's cards have weight 1.0 for observer,
        // 0.0 for others. Unknown cards have weight 1.0 for all non-observer players.
        let mut soft_weights = [[0.0f32; 32]; 4];
        for card in 0..32u8 {
            let bit = card_to_bit(card);
            if observer_hand & bit != 0 {
                soft_weights[observer as usize][card as usize] = 1.0;
            } else {
                for p in 0..4u8 {
                    if p != observer {
                        soft_weights[p as usize][card as usize] = 1.0;
                    }
                }
            }
        }

        BeliefState {
            observer,
            observer_hand,
            soft_weights,
            constraints: Vec::new(),
        }
    }

    /// Check all hard constraints against a set of hands.
    pub fn check_constraints(&self, hands: &[CardSet; 4]) -> bool {
        self.constraints.iter().all(|c| c.is_satisfied(hands))
    }

    /// Get the accumulated constraints (for testing/debug).
    pub fn constraints(&self) -> &[ActionConstraint] {
        &self.constraints
    }
}

    /// Record a bid action by another player. Adds constraints + adjusts soft weights.
    ///
    /// `state` is the state BEFORE the action was applied.
    pub fn record_bid(&mut self, player: u8, action: u8, state: &GameState) {
        use crate::bidding;

        if player == self.observer {
            return; // We know our own hand
        }

        if action == bidding::BID_PASS {
            self.record_pass(player, state);
        } else if action == bidding::BID_COINCHE {
            self.record_coinche(player, state);
        } else if action == bidding::BID_SURCOINCHE {
            // Surcoinche: no strong constraint (it's a response to coinche)
        } else {
            self.record_positive_bid(player, action, state);
        }
    }

    fn record_positive_bid(&mut self, player: u8, action: u8, _state: &GameState) {
        let (bid_value, suit_idx) = bidding::decode_bid(action);
        let suit = Suit::from_u8(suit_idx);
        let min_score = bid_value_to_threshold(bid_value);

        // Hard constraint
        self.constraints.push(ActionConstraint {
            player,
            kind: ConstraintKind::Bid { suit, min_score },
        });

        // Soft weights: boost trump honors for bidder
        let j = make_card(suit, 3); // Jack
        let n = make_card(suit, 2); // Nine
        let a = make_card(suit, 7); // Ace
        let t = make_card(suit, 6); // Ten

        self.apply_factor(player, j, 5.0);
        self.apply_factor(player, n, 3.0);
        self.apply_factor(player, a, 2.0);
        self.apply_factor(player, t, 1.5);

        // Boost remaining trump cards slightly
        for rank in [0, 1, 4, 5] {
            // 7, 8, Q, K
            let card = make_card(suit, rank);
            self.apply_factor(player, card, 1.5);
        }
    }

    fn record_pass(&mut self, player: u8, state: &GameState) {
        let partner = GameState::partner(player);
        let partner_had_bid = state.last_bid_value > 0 && state.last_bidder == partner;
        let active_suit = if state.last_bid_value > 0 {
            Some(Suit::from_u8(state.last_bid_suit))
        } else {
            None
        };

        // Determine overbid value needed
        let min_overbid_value = if state.last_bid_value > 0 {
            state.last_bid_value + 1 // need to bid higher
        } else {
            0 // opening pass
        };

        // Count auction position (how many players have acted this round)
        let auction_position = state.consecutive_passes;

        self.constraints.push(ActionConstraint {
            player,
            kind: ConstraintKind::Pass {
                min_overbid_value,
                auction_position,
                partner_had_bid,
                active_suit,
            },
        });

        // Soft weights: reduce trump honors for passer
        for suit_idx in 0..4u8 {
            let s = Suit::from_u8(suit_idx);
            let j = make_card(s, 3);
            let n = make_card(s, 2);
            self.apply_factor(player, j, 0.6);
            self.apply_factor(player, n, 0.7);
        }
    }

    fn record_coinche(&mut self, player: u8, state: &GameState) {
        // Coinche = defensive holding in opponent's trump
        let trump = Suit::from_u8(state.last_bid_suit);
        let j = make_card(trump, 3);
        let n = make_card(trump, 2);
        self.apply_factor(player, j, 3.0);
        self.apply_factor(player, n, 2.5);

        // Side aces boosted (defensive strength)
        for suit_idx in 0..4u8 {
            if suit_idx != state.last_bid_suit {
                let ace = make_card(Suit::from_u8(suit_idx), 7);
                self.apply_factor(player, ace, 2.0);
            }
        }
    }

    /// Apply multiplicative factor to a single card weight for a player.
    fn apply_factor(&mut self, player: u8, card: Card, factor: f32) {
        if player == self.observer {
            return;
        }
        // Only modify if this card is unknown to observer
        if self.observer_hand & card_to_bit(card) != 0 {
            return;
        }
        self.soft_weights[player as usize][card as usize] *= factor;
    }

```rust
    /// Record a play action. Updates voids, trump ceilings, soft weights.
    pub fn record_play(&mut self, player: u8, card: Card, state: &GameState) {
        // Mark the played card as weight 0 for all players
        for p in 0..4 {
            self.soft_weights[p][card as usize] = 0.0;
        }

        if player == self.observer {
            return;
        }

        let card_s = card_suit_u8(card);
        let trump_suit = state.contract.trump;

        // === Hard constraints from following rules ===
        if state.trick_count > 0 {
            let lead_card = state.current_trick[state.trick_lead as usize];
            let lead_suit = card_suit(lead_card) as u8;

            if card_s != lead_suit {
                // Didn't follow → void in lead suit
                self.mark_void(player, lead_suit);

                if card_s != trump_suit {
                    // Didn't trump either → if must trump, void in trump
                    if !self.partner_is_master(state, player) {
                        self.mark_void(player, trump_suit);
                    }
                }

                // Trump ceiling if cut with undertrump
                if card_s == trump_suit {
                    self.apply_trump_ceiling(state, player, card);
                }
            } else if lead_suit == trump_suit {
                // Following trump — check overtrump constraint
                self.apply_trump_ceiling(state, player, card);
            }
        }

        // === Soft constraints (same patterns as CardBeliefs) ===
        if state.trick_count == 0 {
            // Leader patterns
            let played_rank = card_rank(card);
            if card_s == trump_suit {
                self.apply_suit_factor(player, trump_suit, 1.5);
            } else if played_rank == 7 {
                // Led Ace → likely has 10/K
                let ten = suit_card(card_s, 6);
                let king = suit_card(card_s, 5);
                self.apply_factor(player, ten, 2.0);
                self.apply_factor(player, king, 1.5);
            }
        } else {
            let lead_card = state.current_trick[state.trick_lead as usize];
            let lead_suit = card_suit(lead_card) as u8;
            if card_s != lead_suit && card_s == trump_suit {
                // Cut with low trump → likely lacks stronger
                let played_strength = TRUMP_STRENGTH[card_rank(card) as usize];
                if played_strength <= 2 {
                    for p in 0..4u8 {
                        if p == self.observer || p == player { continue; }
                        let j = suit_card(trump_suit, 3);
                        let n = suit_card(trump_suit, 2);
                        self.apply_factor(p, j, 1.3);
                        self.apply_factor(p, n, 1.3);
                    }
                }
            }
        }
    }

    /// Mark a player as void in a suit (hard constraint).
    fn mark_void(&mut self, player: u8, suit: u8) {
        for rank in 0..8u8 {
            let card = suit * 8 + rank;
            self.soft_weights[player as usize][card as usize] = 0.0;
        }
    }

    /// Apply multiplicative factor to all cards of a suit for a player.
    fn apply_suit_factor(&mut self, player: u8, suit: u8, factor: f32) {
        if player == self.observer { return; }
        for rank in 0..8u8 {
            let card = suit * 8 + rank;
            if self.observer_hand & card_to_bit(card) == 0 {
                self.soft_weights[player as usize][card as usize] *= factor;
            }
        }
    }

    /// Apply trump ceiling: if player couldn't overtrump, they have no stronger trump.
    fn apply_trump_ceiling(&mut self, state: &GameState, player: u8, played_card: Card) {
        let trump_suit = state.contract.trump;
        let played_rank = card_rank(played_card);
        let played_strength = TRUMP_STRENGTH[played_rank as usize];

        // Find the best trump strength already on the trick
        let mut best_strength = 0u8;
        for i in 0..state.trick_count as usize {
            let tc = state.current_trick[(state.trick_lead as usize + i) % 4];
            if card_suit(tc) as u8 == trump_suit {
                let s = TRUMP_STRENGTH[card_rank(tc) as usize];
                if s > best_strength {
                    best_strength = s;
                }
            }
        }

        if played_strength < best_strength {
            // Player couldn't overtrump → zero out all trump ranks stronger than played
            for rank in 0..8u8 {
                if TRUMP_STRENGTH[rank as usize] > played_strength {
                    let card = trump_suit * 8 + rank;
                    self.soft_weights[player as usize][card as usize] = 0.0;
                }
            }
        }
    }

    /// Check if player's partner is currently winning the trick.
    fn partner_is_master(&self, state: &GameState, player: u8) -> bool {
        if state.trick_count == 0 { return false; }
        let partner = GameState::partner(player);
        // Simplified: check if partner played the current winning card
        let trump_suit = state.contract.trump;
        let lead_suit = card_suit(state.current_trick[state.trick_lead as usize]) as u8;

        let mut best_seat = state.trick_lead;
        let mut best_strength = 0u16;

        for i in 0..state.trick_count as usize {
            let seat = ((state.trick_lead as usize + i) % 4) as u8;
            let c = state.current_trick[seat as usize];
            let cs = card_suit(c) as u8;

            let strength = if cs == trump_suit {
                100 + TRUMP_STRENGTH[card_rank(c) as usize] as u16
            } else if cs == lead_suit {
                card_rank(c) as u16
            } else {
                0 // off-suit, can't win
            };

            if strength > best_strength {
                best_strength = strength;
                best_seat = seat;
            }
        }

        best_seat == partner
    }
```

    /// Generate a determinization consistent with all accumulated beliefs.
    /// Uses weighted sampling (soft biases) + hard rejection (constraints).
    /// Falls back to unfiltered determinization if all attempts fail.
    pub fn determinize(
        &self,
        state: &GameState,
        rng: &mut impl Rng,
    ) -> Option<GameState> {
        use crate::determinize::{determinize_greedy, determinize_weighted};

        let has_constraints = !self.constraints.is_empty();

        for _ in 0..500 {
            // Generate candidate with soft weight bias
            let candidate = determinize_weighted(state, self.observer, &self.soft_weights, rng)
                .or_else(|| determinize_greedy(state, self.observer, rng));

            let candidate = match candidate {
                Some(c) => c,
                None => continue,
            };

            // Check hard constraints
            if !has_constraints || self.check_constraints(&candidate.hands) {
                return Some(candidate);
            }
        }

        // Fallback: return unfiltered determinization
        determinize_greedy(state, self.observer, rng)
    }

    }

    pub fn observe(&mut self, player: u8, action: u8, state: &GameState) {

            Some(b) => b,
            None => return,

            Phase::Bidding => belief.record_bid(player, action, state),
            Phase::Playing => belief.record_play(player, action as Card, state),

impl BisDdAgent {
    /// Expected value of bidding `bid_value` (encoded) in `suit`.
    fn ev_bid(results: &[[u8; 4]], suit: u8, bid_value: u8, team: u8) -> f32 {
        let contract = bid_value as i32 * 10;
        let n = results.len() as f32;
        let mut total = 0.0f32;

        for r in results {
            let ns_pts = r[suit as usize] as i32;
            let team_pts = if team == 0 { ns_pts } else { 162 - ns_pts };

            if team_pts >= contract {
                total += (team_pts + contract) as f32;
            } else {
                total -= contract as f32;
            }
        }

        total / n
    }

    /// Expected value of passing (letting current contract stand).
    fn ev_pass(results: &[[u8; 4]], state: &GameState, team: u8) -> f32 {
        if state.last_bid_value == 0 {
            return 0.0;
        }

        let bid_suit = state.last_bid_suit;
        let bidder_team = GameState::player_team(state.last_bidder);
        let contract = state.last_bid_value as i32 * 10;
        let coinche_mult = match state.coinche_state {
            0 => 1i32,
            1 => 2,
            _ => 4,
        };

        let n = results.len() as f32;
        let mut total = 0.0f32;

        for r in results {
            let ns_pts = r[bid_suit as usize] as i32;
            let bidder_pts = if bidder_team == 0 { ns_pts } else { 162 - ns_pts };

            if bidder_pts >= contract {
                let score = (bidder_pts + contract) * coinche_mult;
                if bidder_team == team {
                    total += score as f32;
                } else {
                    total -= score as f32;
                }
            } else {
                let score = contract * coinche_mult;
                if bidder_team == team {
                    total -= score as f32;
                } else {
                    total += score as f32;
                }
            }
        }

        total / n
    }

    /// Expected value of coinching opponent's contract.
    fn ev_coinche(results: &[[u8; 4]], state: &GameState, team: u8) -> f32 {
        let bid_suit = state.last_bid_suit;
        let opp_team = 1 - team;
        let contract = state.last_bid_value as i32 * 10;
        let coinche_mult = 2i32;

        let n = results.len() as f32;
        let mut total = 0.0f32;

        for r in results {
            let ns_pts = r[bid_suit as usize] as i32;
            let opp_pts = if opp_team == 0 { ns_pts } else { 162 - ns_pts };

            if opp_pts >= contract {
                total -= ((opp_pts + contract) * coinche_mult) as f32;
            } else {
                total += (contract * coinche_mult) as f32;
            }
        }

        total / n
    }

            let team_pts = if team == 0 { ns_pts } else { 162 - ns_pts };

            if team_pts >= 252 {
                total += (252 + 250) as f32;
            } else {
                total -= 250.0;
            }
        }

        total / n
    }

impl BisDdAgent {
    fn decide_bid(&mut self, state: &GameState) -> u8 {
        let belief = match &self.belief {
            Some(b) => b,
            None => return BID_PASS,
        };

        let player = state.current_player();
        let team = GameState::player_team(player);
        let hand = state.hands[player as usize];
        let legal = state.legal_actions();

        // 1. Prefilter candidate suits
        let candidates: Vec<u8> = (0..4u8)
            .filter(|&s| {
                evaluate_for_trump(hand, Suit::from_u8(s)) >= self.config.prefilter_threshold
            })
            .collect();

        // 2. Determine suits to solve (candidates + opponent's suit if bid exists)
        let mut suits_to_solve: Vec<u8> = candidates.clone();
        if state.last_bid_value > 0 && !suits_to_solve.contains(&state.last_bid_suit) {
            suits_to_solve.push(state.last_bid_suit);
        }

        // If nothing to evaluate, pass
        if suits_to_solve.is_empty() {
            return BID_PASS;
        }

        // 3. Generate determinizations (time-bounded, min_dets minimum)
        let deadline = Instant::now() + Duration::from_millis(self.config.bid_time_ms as u64);
        let mut results: Vec<[u8; 4]> = Vec::with_capacity(self.config.min_dets as usize * 2);
        let mut attempts = 0u32;

        while (results.len() < self.config.min_dets as usize || Instant::now() < deadline)
            && attempts < self.config.min_dets * 10
        {
            attempts += 1;
            if let Some(det) = belief.determinize(state, &mut self.rng) {
                let mut pts = [0u8; 4];
                for &s in &suits_to_solve {
                    pts[s as usize] =
                        solve_for_trump_reuse_tt(det.hands, state.dealer, s, &mut self.tt_buf)[0];
                }
                results.push(pts);
            }

            // Respect time budget after min_dets
            if results.len() >= self.config.min_dets as usize && Instant::now() >= deadline {
                break;
            }
        }

        if results.is_empty() {
            return BID_PASS;
        }

        // 4. Evaluate all candidate actions
        let mut best_action = BID_PASS;
        let mut best_ev = Self::ev_pass(&results, state, team);

        // Evaluate suit × value combinations
        for &suit in &candidates {
            // Find legal bid values for this suit
            for bid_value in 8u8..=16 {
                let action = bidding::encode_bid(bid_value, suit);
                if legal & (1u64 << action) == 0 {
                    continue;
                }
                let ev = Self::ev_bid(&results, suit, bid_value, team);
                if ev > best_ev {
                    best_ev = ev;
                    best_action = action;
                }
            }
        }

        // Evaluate capot
        if self.config.evaluate_capot {
            for &suit in &candidates {
                let action = 37 + suit;
                if legal & (1u64 << action) != 0 {
                    let ev = Self::ev_capot(&results, suit, team);
                    if ev > best_ev {
                        best_ev = ev;
                        best_action = action;
                    }
                }
            }
        }

        // Evaluate coinche
        if legal & (1u64 << bidding::BID_COINCHE) != 0 {
            let ev = Self::ev_coinche(&results, state, team);
            if ev > best_ev {
                best_ev = ev;
                best_action = bidding::BID_COINCHE;
            }
        }

        best_action
    }
}

impl BisDdAgent {
    fn decide_play(&mut self, state: &GameState) -> u8 {
        let belief = match &self.belief {
            Some(b) => b,
            None => {
                // No beliefs → use first legal action
                return state.legal_actions().trailing_zeros() as u8;
            }
        };

        let player = state.current_player();
        let team = GameState::player_team(player);
        let legal = state.legal_actions();
        let maximizing = team == 0; // NS maximizes, EW minimizes

        // Score accumulators
        let mut score_sum = [0i64; 32];
        let mut score_count = [0u32; 32];

        // Time-bounded determinization
        let cards_left = card_count(state.hands[player as usize]);
        let scaled_ms = (self.config.play_time_ms as u64 * cards_left as u64) / 8;
        let deadline = Instant::now() + Duration::from_millis(scaled_ms.max(1));
        let mut successful_dets = 0u32;
        let mut attempts = 0u32;

        while (successful_dets < self.config.min_dets || Instant::now() < deadline)
            && attempts < self.config.min_dets * 10
        {
            attempts += 1;
            let det = match belief.determinize(state, &mut self.rng) {
                Some(d) => d,
                None => continue,
            };

            let scores = solve_with_scores(&det, Some(&mut self.tt_buf));
            for i in 0..scores.count {
                let (card, ns_pts) = scores.scores[i];
                score_sum[card as usize] += ns_pts as i64;
                score_count[card as usize] += 1;
            }
            successful_dets += 1;

            if successful_dets >= self.config.min_dets && Instant::now() >= deadline {
                break;
            }
        }

        // Pick best card
        let mut best_card = legal.trailing_zeros() as u8;
        let mut best_avg: f32 = if maximizing {
            f32::NEG_INFINITY
        } else {
            f32::INFINITY
        };

        let mut mask = legal;
        while mask != 0 {
            let card = mask.trailing_zeros() as u8;
            let count = score_count[card as usize];
            let avg = if count > 0 {
                score_sum[card as usize] as f32 / count as f32
            } else {
                81.0 // neutral fallback
            };

            let dominated = if maximizing {
                avg > best_avg
            } else {
                avg < best_avg
            };
            if dominated {
                best_avg = avg;
                best_card = card;
            }
            mask &= mask - 1;
        }

        best_card
    }
}
